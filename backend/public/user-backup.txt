  function playNotificationSound() {
    try {
      const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const oscillator = audioCtx.createOscillator();
      const gainNode = audioCtx.createGain();
      oscillator.connect(gainNode);
      gainNode.connect(audioCtx.destination);
      oscillator.frequency.value = 880;
      oscillator.type = 'sine';
      gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);
      gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.5);
      oscillator.start(audioCtx.currentTime);
      oscillator.stop(audioCtx.currentTime + 0.5);
      setTimeout(() => {
        const osc2 = audioCtx.createOscillator();
        const gain2 = audioCtx.createGain();
        osc2.connect(gain2);
        gain2.connect(audioCtx.destination);
        osc2.frequency.value = 1100;
        osc2.type = 'sine';
        gain2.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain2.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.5);
        osc2.start(audioCtx.currentTime);
        osc2.stop(audioCtx.currentTime + 0.5);
      }, 200);
    } catch (e) {}
  }

  function sendDesktopNotification(title, message, icon = '💻') {
    if ('Notification' in window && Notification.permission === 'granted') {
      try {
        const notification = new Notification(icon + ' ' + title, {
          body: message,
          icon: 'data:image/svg+xml,' + encodeURIComponent(`<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><text y=".9em" font-size="90">💻</text></svg>`),
          tag: 'billing-notification',
          requireInteraction: true
        });
        notification.onclick = function() {
          window.focus();
          notification.close();
        };
        setTimeout(() => notification.close(), 30000);
      } catch (e) {}
    }
  }

  function showForceModal(toast) {
    const iconMap = {
      'danger': 'bi-exclamation-triangle-fill',
      'success': 'bi-check-circle-fill',
      'warn': 'bi-pause-circle-fill',
      'info': 'bi-info-circle-fill'
    };
    document.getElementById('forceModalIcon').className = 'bi ' + (iconMap[toast.variant] || 'bi-info-circle-fill');
    document.getElementById('forceModalTitle').innerText = toast.title || 'Notifikasi';
    document.getElementById('forceModalMessage').innerText = toast.message || '';
    
    const modal = document.getElementById('forceModal');
    modal.className = 'force-modal ' + (toast.variant || 'info');
    
    document.getElementById('forceModalOverlay').classList.add('show');
    playNotificationSound();
  }

  function closeForceModal() {
    document.getElementById('forceModalOverlay').classList.remove('show');
    if (isElectron && window.electronAPI) {
      window.electronAPI.dismissNotification();
    }
    if (pendingForceModals.length > 0) {
      setTimeout(() => {
        const next = pendingForceModals.shift();
        showForceModal(next);
      }, 400);
    }
  }

  function triggerNotification(toast) {
    playNotificationSound();
    const iconMap = { 'danger': '⚠️', 'success': '✅', 'warn': '⏸️', 'info': 'ℹ️' };
    sendDesktopNotification(
      toast.title || 'Notifikasi', 
      toast.message || '', 
      iconMap[toast.variant] || 'ℹ️'
    );
    
    if (isElectron) {
      showForceModal(toast);
      if (window.electronAPI) {
        window.electronAPI.triggerNotification({ title: toast.title, message: toast.message });
      }
      return;
    }
    
    if (document.hidden) {
      pendingForceModals.push(toast);
      try {
        window.focus();
        if (window.parent && window.parent !== window) {
          window.parent.focus();
        }
      } catch (e) {}
      startTitleBlink(toast.title || 'Pesan Baru');
      changeFavicon(true);
    } else {
      queueToast(toast);
    }
  }

      if (pendingForceModals.length > 0) {
        const firstModal = pendingForceModals.shift();
        setTimeout(() => showForceModal(firstModal), 300);
      }
    }
  });

  function showNextWarning() {
    if (warnQueue.length === 0) { warnTimer = null; return; }
    const t = warnQueue.shift();
    document.getElementById('warnMsg').innerText = t.message || '';
    document.getElementById('warnTitle').innerText = t.title || '';
    document.getElementById('warnIcon').className = 'bi ' + (t.icon || 'bi-info-circle-fill');
    const card = document.getElementById('warnCard');
    card.className = 'warn-card' + (t.variant && t.variant !== 'danger' ? ' variant-' + t.variant : '');
    const bar = document.getElementById('warnBar');
    bar.style.animation = 'none'; void bar.offsetWidth; bar.style.animation = null;
    document.getElementById('warnBanner').classList.add('show');
    const duration = t.duration || 20000;
    warnTimer = setTimeout(dismissWarning, duration);

  function parseServerTime(raw) {
    if (!raw) return '00:00:00';
    const parts = String(raw).split('.');
    if (parts.length !== 3) return String(raw);
    const [h, m, s] = parts;
    return `${h.padStart(2, '0')}:${m.padStart(2, '0')}:${s.padStart(2, '0')}`;
  }

  function detectStatusTransition(newStatus) {
    const prev = lastKnownStatus;
    lastKnownStatus = newStatus;
    if (prev === undefined || prev === newStatus) return false;

    if (prev === 'available' && newStatus === 'in-use') {
      triggerNotification({ variant: 'success', icon: 'bi-play-circle-fill', title: '▶️ SESI DIMULAI', message: 'Petugas telah memulai sesi Anda di unit ini.', duration: 8000 });
    } else if ((prev === 'in-use' || prev === 'paused') && newStatus === 'available') {
      triggerNotification({ variant: 'info', icon: 'bi-stop-circle-fill', title: '⏹️ SESI SELESAI', message: 'Sesi Anda telah diakhiri oleh petugas.', duration: 8000 });
    } else if (prev === 'in-use' && newStatus === 'paused') {
      triggerNotification({ variant: 'warn', icon: 'bi-pause-circle-fill', title: '⏸️ SESI DIJEDA', message: 'Petugas menjeda sesi Anda. Saldo waktu berhenti berkurang.', duration: 8000 });
    } else if (prev === 'paused' && newStatus === 'in-use') {
      triggerNotification({ variant: 'success', icon: 'bi-play-circle-fill', title: '▶️ SESI DILANJUTKAN', message: 'Sesi Anda dilanjutkan oleh petugas.', duration: 8000 });
    }
    return true;
  }

  function detectDurationChange(pc, statusChanged) {
    const curDur = (pc.current_duration_minutes !== undefined && pc.current_duration_minutes !== null)
      ? Number(pc.current_duration_minutes) : null;

    if (!statusChanged && (pc.status === 'in-use' || pc.status === 'paused') &&
        curDur !== null && lastKnownDurationMinutes !== undefined && lastKnownDurationMinutes !== null) {
      const diff = curDur - lastKnownDurationMinutes;
      if (diff > 0) {
        triggerNotification({ variant: 'success', icon: 'bi-plus-circle-fill', title: '➕ WAKTU DITAMBAHKAN', message: `+${diff} menit ditambahkan oleh petugas ke saldo Anda.` });
      } else if (diff < 0) {
        triggerNotification({ variant: 'warn', icon: 'bi-dash-circle-fill', title: '➖ WAKTU DIKURANGI', message: `-${Math.abs(diff)} menit dikurangi oleh petugas.` });
      }
    }
    lastKnownDurationMinutes = curDur;
  }

  async function popOutWidget() {
    if (!('documentPictureInPicture' in window)) {
      alert('Fitur pop-out hanya tersedia di Chrome 116+.\nUpdate Chrome Anda untuk menggunakan fitur ini.');
      return;
    }

    if (pipWindow && !pipWindow.closed) {
      pipWindow.focus();
      return;
    }

    const pc = myPC();
    if (!pc) {
      alert('Belum ada PC yang dipilih.');
      return;
    }

    try {
      // Buka PiP window dengan ukuran besar
      pipWindow = await documentPictureInPicture.requestWindow({
        width: 420,
        height: 480,
      });

      // Copy semua stylesheet dari halaman utama
      [...document.styleSheets].forEach((styleSheet) => {
        try {
          const cssRules = [...styleSheet.cssRules].map(rule => rule.cssText).join('');
          const style = document.createElement('style');
          style.textContent = cssRules;
          pipWindow.document.head.appendChild(style);
        } catch (e) {
          const link = document.createElement('link');
          link.rel = 'stylesheet';
          link.type = styleSheet.type;
          link.media = styleSheet.media;
          link.href = styleSheet.href;
          pipWindow.document.head.appendChild(link);
        }
      });

      // Tambahkan font & icons
      const fonts = pipWindow.document.createElement('link');
      fonts.rel = 'stylesheet';
      fonts.href = 'https://fonts.googleapis.com/css2?family=Fraunces:opsz,wght@9..144,400;9..144,500;9..144,600;9..144,700;9..144,900&family=Plus+Jakarta+Sans:wght@400;500;600;700&family=JetBrains+Mono:wght@500;600;700&display=swap';
      pipWindow.document.head.appendChild(fonts);

      const icons = pipWindow.document.createElement('link');
      icons.rel = 'stylesheet';
      icons.href = 'https://cdnjs.cloudflare.com/ajax/libs/bootstrap-icons/1.11.3/font/bootstrap-icons.min.css';
      pipWindow.document.head.appendChild(icons);

      // Build konten PiP
      buildPipContent();

      pipWindow.addEventListener('pagehide', () => {
        pipWindow = null;
      });

      console.log('Pop-out aktif. Widget akan selalu di atas.');
    } catch (e) {
      console.error('Pop-out error:', e);
      alert('Gagal membuka pop-out: ' + e.message);
    }
  }
